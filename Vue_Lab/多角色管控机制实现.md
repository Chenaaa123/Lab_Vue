# 校园实验室预约系统 - 多角色管控机制实现

## 一、多角色管控机制概述

### 1.1 角色定义
本系统采用RBAC（基于角色的访问控制）模型，定义了4种用户角色：

| 角色名称 | 角色代码 | 权限级别 | 主要职责 |
|----------|----------|----------|----------|
| 系统管理员 | SYSTEM_ADMIN | 最高权限 | 系统配置、用户管理、全局数据管理 |
| 实验室管理员 | LAB_MANAGER | 中级权限 | 实验室管理、预约审核、设备维护 |
| 教师 | TEACHER | 普通权限 | 预约实验室、查看统计、指导学生 |
| 学生 | STUDENT | 基础权限 | 预约实验室、查看公告、提交报修 |

### 1.2 权限层级结构
```
系统管理员 (最高权限)
├── 用户管理 (增删改查所有用户)
├── 实验室管理 (全局实验室配置)
├── 系统配置 (参数设置、权限配置)
└── 数据统计 (全局数据分析)

实验室管理员 (中级权限)
├── 实验室管理 (管理负责的实验室)
├── 预约审核 (审核相关预约)
├── 设备维护 (设备报修处理)
└── 数据统计 (负责实验室的数据分析)

教师 (普通权限)
├── 预约管理 (预约和使用实验室)
├── 学生指导 (查看学生预约情况)
├── 报修申请 (提交设备报修)
└── 数据查看 (查看相关统计数据)

学生 (基础权限)
├── 预约申请 (预约实验室)
├── 公告查看 (查看系统公告)
├── 报修申请 (提交设备报修)
└── 个人信息 (管理个人信息)
```

### 2.3 自定义权限注解

#### 权限检查注解
```java
@Target({ElementType.METHOD, ElementType.TYPE})
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface RequirePermission {
    /**
     * 需要的权限代码
     */
    String[] value() default {};
    
    /**
     * 实验室管理员特殊权限检查
     */
    String labManager() default "";
    
    /**
     * 是否检查实验室所有权
     */
    boolean checkLabOwnership() default false;
    
    /**
     * 权限检查模式
     */
    Logical logical() default Logical.AND;
}

public enum Logical {
    AND, OR
}
```

#### 权限检查切面
```java
@Aspect
@Component
@Order(1)
public class PermissionAspect {
    
    @Autowired
    private PermissionService permissionService;
    
    @Autowired
    private UserService userService;
    
    @Autowired
    private LabService labService;
    
    @Around("@annotation(requirePermission)")
    public Object checkPermission(ProceedingJoinPoint joinPoint, RequirePermission requirePermission) throws Throwable {
        // 获取当前用户
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !authentication.isAuthenticated()) {
            throw new AccessDeniedException("用户未登录");
        }
        
        UserDetailsImpl userDetails = (UserDetailsImpl) authentication.getPrincipal();
        User currentUser = userDetails.getUser();
        
        // 检查基础权限
        if (requirePermission.value().length > 0) {
            boolean hasPermission = checkPermissions(currentUser, requirePermission.value(), requirePermission.logical());
            if (!hasPermission) {
                throw new AccessDeniedException("权限不足");
            }
        }
        
        // 实验室管理员特殊权限检查
        if (!StringUtils.isEmpty(requirePermission.labManager())) {
            boolean hasLabPermission = checkLabManagerPermission(currentUser, requirePermission.labManager());
            if (!hasLabPermission) {
                throw new AccessDeniedException("实验室管理员权限不足");
            }
        }
        
        // 检查实验室所有权
        if (requirePermission.checkLabOwnership()) {
            boolean hasOwnership = checkLabOwnership(currentUser, joinPoint.getArgs());
            if (!hasOwnership) {
                throw new AccessDeniedException("您不是该实验室的管理员");
            }
        }
        
        return joinPoint.proceed();
    }
    
    private boolean checkPermissions(User user, String[] requiredPermissions, Logical logical) {
        Set<String> userPermissions = permissionService.getUserPermissions(user.getId());
        
        if (logical == Logical.AND) {
            // 需要拥有所有权限
            return Arrays.stream(requiredPermissions)
                    .allMatch(userPermissions::contains);
        } else {
            // 只需要拥有其中一个权限
            return Arrays.stream(requiredPermissions)
                    .anyMatch(userPermissions::contains);
        }
    }
    
    private boolean checkLabManagerPermission(User user, String permission) {
        if (user.getRole() != UserRole.LAB_MANAGER) {
            return false;
        }
        
        Set<String> labManagerPermissions = permissionService.getLabManagerPermissions(user.getId());
        return labManagerPermissions.contains(permission);
    }
    
    private boolean checkLabOwnership(User user, Object[] args) {
        if (user.getRole() != UserRole.LAB_MANAGER) {
            return false;
        }
        
        // 从方法参数中提取实验室ID
        Long labId = extractLabIdFromArgs(args);
        if (labId == null) {
            return false;
        }
        
        // 检查用户是否是该实验室的管理员
        return labService.isLabManager(user.getId(), labId);
    }
    
    private Long extractLabIdFromArgs(Object[] args) {
        for (Object arg : args) {
            if (arg instanceof Long) {
                return (Long) arg;
            } else if (arg instanceof Reservation) {
                return ((Reservation) arg).getLab().getId();
            } else if (arg instanceof Lab) {
                return ((Lab) arg).getId();
            }
        }
        return null;
    }
}
```

### 2.4 权限服务实现

#### 权限服务接口
```java
@Service
public class PermissionService {
    
    @Autowired
    private RolePermissionRepository rolePermissionRepository;
    
    @Autowired
    private LabManagerRepository labManagerRepository;
    
    /**
     * 获取用户所有权限
     */
    public Set<String> getUserPermissions(Long userId) {
        User user = userService.findById(userId);
        Set<String> permissions = new HashSet<>();
        
        // 根据角色获取基础权限
        permissions.addAll(getRolePermissions(user.getRole()));
        
        // 如果是实验室管理员，添加特殊权限
        if (user.getRole() == UserRole.LAB_MANAGER) {
            permissions.addAll(getLabManagerPermissions(userId));
        }
        
        return permissions;
    }
    
    /**
     * 根据角色获取权限
     */
    private Set<String> getRolePermissions(UserRole role) {
        return rolePermissionRepository.findByRoleCode(role.name())
                .stream()
                .map(rp -> rp.getPermission().getPermissionCode())
                .collect(Collectors.toSet());
    }
    
    /**
     * 获取实验室管理员特殊权限
     */
    public Set<String> getLabManagerPermissions(Long userId) {
        Set<String> permissions = new HashSet<>();
        
        // 获取用户管理的实验室
        List<LabManagerAssignment> assignments = labManagerRepository.findByUserId(userId);
        
        for (LabManagerAssignment assignment : assignments) {
            // 添加实验室管理权限
            permissions.add("LAB:MANAGE:" + assignment.getLab().getId());
            permissions.add("RESERVATION:AUDIT:" + assignment.getLab().getId());
            permissions.add("REPAIR:PROCESS:" + assignment.getLab().getId());
        }
        
        return permissions;
    }
    
    /**
     * 检查用户是否有指定权限
     */
    public boolean hasPermission(Long userId, String permission) {
        Set<String> userPermissions = getUserPermissions(userId);
        return userPermissions.contains(permission);
    }
    
    /**
     * 检查用户是否是实验室管理员
     */
    public boolean isLabManager(Long userId) {
        User user = userService.findById(userId);
        return user.getRole() == UserRole.LAB_MANAGER;
    }
    
    /**
     * 检查用户是否管理指定实验室
     */
    public boolean managesLab(Long userId, Long labId) {
        return labManagerRepository.existsByUserIdAndLabIdAndStatus(userId, labId, "ACTIVE");
    }
}
```

```javascript
    }
  }
]

const router = createRouter({
  history: createWebHistory(),
  routes
})

// 全局路由守卫
router.beforeEach((to, from, next) => {
  const isAuthenticated = store.getters.isAuthenticated
  const userRole = store.getters.userRole
  const userPermissions = store.getters.userPermissions

  // 检查是否需要登录
  if (to.meta.requiresAuth && !isAuthenticated) {
    next('/login')
    return
  }

  // 检查角色权限
  if (to.meta.roles && !to.meta.roles.includes(userRole)) {
    next('/403')
    return
  }

  // 检查具体权限
  if (to.meta.permissions) {
    const hasPermission = to.meta.permissions.some(permission => 
      userPermissions.includes(permission)
    )
    if (!hasPermission) {
      next('/403')
      return
    }
  }

  next()
})

export default router
```

#### 权限指令

```javascript
// directives/permission.js
import store from '@/store'

const permission = {
  mounted(el, binding) {
    const { value } = binding
    const userPermissions = store.getters.userPermissions

    if (value && value.length > 0) {
      const hasPermission = value.some(permission => 
        userPermissions.includes(permission)
      )
      
      if (!hasPermission) {
        el.parentNode && el.parentNode.removeChild(el)
      }
    }
  },
  
  updated(el, binding) {
    const { value } = binding
    const userPermissions = store.getters.userPermissions

    if (value && value.length > 0) {
      const hasPermission = value.some(permission => 
        userPermissions.includes(permission)
      )
      
      if (!hasPermission) {
        el.style.display = 'none'
      } else {
        el.style.display = ''
      }
    }
  }
}

export default permission
```

### 3.3 Vuex状态管理

```javascript
// store/modules/permission.js
import { getUserPermissions } from '@/api/auth'

const state = {
  userPermissions: [],
  rolePermissions: {},
  labPermissions: []
}

const mutations = {

const actions = {

      return response.data
    } catch (error) {

    }
  },

  }
}

const getters = {
  userPermissions: state => state.userPermissions,
  hasPermission: (state) => (permission) => {
    return state.userPermissions.includes(permission)
  },
  hasAnyPermission: (state) => (permissions) => {
    return permissions.some(permission => 
      state.userPermissions.includes(permission)
    )
  },
  hasAllPermissions: (state) => (permissions) => {
    return permissions.every(permission => 
      state.userPermissions.includes(permission)
    )
  }
}
```

```javascript
// utils/request.js
import axios from 'axios'
import { ElMessage } from 'element-plus'
import store from '@/store'
import router from '@/router'

const service = axios.create({
  baseURL: process.env.VUE_APP_API_BASE_URL,
  timeout: 10000
})

// 请求拦截器
service.interceptors.request.use(
  config => {
    // 添加认证Token
    const token = localStorage.getItem('token')
    if (token) {
      config.headers['Authorization'] = `Bearer ${token}`
    }
    
    // 添加用户角色信息（可选）
    const userRole = store.getters.userRole
    if (userRole) {
      config.headers['X-User-Role'] = userRole
    }
    
    return config
  },
  error => {
    console.error('请求错误:', error)
    return Promise.reject(error)
  }
)

// 响应拦截器
service.interceptors.response.use(
  response => {
    return response.data
  },
  error => {
    const { response } = error
    
    if (response) {
      switch (response.status) {
        case 401:
          // 未授权，跳转登录页
          ElMessage.error('登录已过期，请重新登录')
          store.dispatch('user/logout')
          router.push('/login')
          break
        case 403:
          // 权限不足
          ElMessage.error('权限不足，无法访问该资源')
          router.push('/403')
          break
        case 500:
          ElMessage.error('服务器错误，请稍后重试')
          break
        default:
          ElMessage.error(response.data?.message || '请求失败')
      }
    } else {
      ElMessage.error('网络错误，请检查网络连接')
    }
    
    return Promise.reject(error)
  }
)

export default service
```

```java
        assignPermissionsToRole(UserRole.TEACHER, getTeacherPermissions());
        assignPermissionsToRole(UserRole.STUDENT, getStudentPermissions());

                rolePermission.setRoleCode(role.name());

                rolePermissionRepository.save(rolePermission);
            }
```
